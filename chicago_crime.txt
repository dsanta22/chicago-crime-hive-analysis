USE dsanta22;



#Step 4: Create RAW table 

CREATE EXTERNAL TABLE IF NOT EXISTS crimes_raw (
  id STRING,
  case_number STRING,
  date_str STRING,
  block STRING,
  iucr STRING,
  primary_type STRING,
  description STRING,
  location_description STRING,
  arrest STRING,
  domestic STRING,
  beat STRING,
  district STRING,
  ward STRING,
  community_area STRING,
  fbi_code STRING,
  x_coordinate STRING,
  y_coordinate STRING,
  year STRING,
  updated_on STRING,
  latitude STRING,
  longitude STRING,
  location STRING
)
ROW FORMAT SERDE 'org.apache.hadoop.hive.serde2.OpenCSVSerde'
WITH SERDEPROPERTIES (
  "separatorChar" = ",",
  "quoteChar" = "\"",
  "escapeChar" = "\\"
)
LOCATION '/user/dsanta22/city_crime_chicago';



---Step 5: Verify table structure

DESCRIBE crimes_raw;


Step #6: Verify the raw table loaded


SELECT COUNT(*) FROM crimes_raw;



STEP #7: Preview important fields 

SELECT
  latitude,
  longitude,
  primary_type,
  date_str
FROM crimes_raw
WHERE latitude IS NOT NULL
  AND longitude IS NOT NULL
LIMIT 20;

--STEP #8: Run the query that created the final map data:

SELECT
  latitude,
  longitude,
  primary_type,
  date_str,
  COUNT(*) AS crime_count
FROM crimes_raw
WHERE latitude IS NOT NULL
  AND longitude IS NOT NULL
GROUP BY
  latitude,
  longitude,
  primary_type,
  date_str;
