#Step 4: Create RAW table 

CREATE EXTERNAL TABLE IF NOT EXISTS crimes_raw (
  id STRING,
  case_number STRING,
  date_str STRING,
  block STRING,
  iucr STRING,
  primary_type STRING,
  description STRING,
  location_description STRING,
  arrest STRING,
  domestic STRING,
  beat STRING,
  district STRING,
  ward STRING,
  community_area STRING,
  fbi_code STRING,
  x_coordinate STRING,
  y_coordinate STRING,
  year STRING,
  updated_on STRING,
  latitude STRING,
  longitude STRING,
  location STRING
)
ROW FORMAT SERDE 'org.apache.hadoop.hive.serde2.OpenCSVSerde'
WITH SERDEPROPERTIES (
  "separatorChar" = ",",
  "quoteChar" = "\"",
  "escapeChar" = "\\"
)
LOCATION '/user/dsanta22/city_crime_chicago';

--STEP #8: Top 5 Crime Types

SELECT
  primary_type,
  COUNT(*) AS total_reported_crimes
FROM crimes_raw
WHERE primary_type IS NOT NULL
GROUP BY primary_type
ORDER BY total_reported_crimes DESC
LIMIT 5;

---8.2 Crime Trends Over Time

SELECT
  year,
  COUNT(*) AS total_crimes
FROM crimes_raw
WHERE year IS NOT NULL
GROUP BY year
ORDER BY year;

---8.3 Crime Hotspots in Chicago

SELECT
  latitude,
  longitude,
  COUNT(*) AS number_of_crimes
FROM crimes_raw
WHERE latitude IS NOT NULL
  AND longitude IS NOT NULL
GROUP BY latitude, longitude;

----8.4 Crime Frequency by Time of Day

SELECT
  CASE
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 6 AND 11 THEN 'Morning'
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 12 AND 17 THEN 'Afternoon'
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 18 AND 23 THEN 'Evening'
    ELSE 'Late Night'
  END AS time_of_day,
  COUNT(*) AS number_of_crimes
FROM crimes_raw
WHERE date_str IS NOT NULL
GROUP BY
  CASE
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 6 AND 11 THEN 'Morning'
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 12 AND 17 THEN 'Afternoon'
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 18 AND 23 THEN 'Evening'
    ELSE 'Late Night'
  END;

----8.5 Crime Type Distribution by Time of Day


SELECT
  CASE
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 6 AND 11 THEN 'Morning'
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 12 AND 17 THEN 'Afternoon'
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 18 AND 23 THEN 'Evening'
    ELSE 'Late Night'
  END AS time_of_day,
  primary_type,
  COUNT(*) AS number_of_crimes
FROM crimes_raw
WHERE date_str IS NOT NULL
  AND primary_type IS NOT NULL
GROUP BY
  CASE
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 6 AND 11 THEN 'Morning'
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 12 AND 17 THEN 'Afternoon'
    WHEN CAST(SUBSTR(date_str, 12, 2) AS INT) BETWEEN 18 AND 23 THEN 'Evening'
    ELSE 'Late Night'
  END,
  primary_type;
